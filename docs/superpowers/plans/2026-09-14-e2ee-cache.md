# E2EE (X3DH + Double Ratchet) + IndexedDB cache Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship full X3DH + Double Ratchet E2EE with multidevice plus an IndexedDB read-cache with delta-sync for chat messages.

**Architecture:** Contracts first in `libs/common` (zod); `chat-service` stores only ciphertext + public key material (devices, prekey bundles, sender-key shares); Gateway is a thin proxy + socket fan-out per recipient device; frontend gets a shared-layer `@org/crypto-e2ee` package (WebCrypto, non-extractable keys) and an IndexedDB cache inside `@org/entities-message`.

**Tech Stack:** NestJS + Prisma (chat-service), WebCrypto AES-GCM + X25519/HKDF (frontend, no new heavy deps unless split into own lazy package), `idb` for IndexedDB, Socket.IO fan-out, Jest (backend) + Vitest (client).

**Spec:** `docs/superpowers/specs/2026-09-14-e2ee-cache-design.md`

## Global Constraints

- Contracts first: new shapes start as zod schemas in `libs/common`; backend wraps them (`createZodDto`), frontend extends them. No hardcoded routes — use `API_ROUTES`. No forked schema copies.
- Apps are shells: all code lives in `libs/`; `apps/` only import and wire.
- Backend transport split: HTTP only in Gateway; lib controllers use `@MessagePattern`/`@EventPattern`. New patterns go in `libs/backend/core/src/constants/queues/*`.
- Guards order: Throttler → Session/Jwt → ActiveAccount → Roles. Request data via `@CurrentUser()` / `@ClientMetadata()`.
- Errors: services throw RPC errors `{ message, status }`; Gateway maps to `HttpException`.
- One service, one database: E2EE tables live in the chat-service Prisma schema only.
- No secrets in logs: only presence flags (`hasDeviceId: !!deviceId`) and `eventType` step logs.
- FSD: slices import only from layers below; across slices via public API `index.ts` only. New packages are app-agnostic.
- Heavy deps (>30 KB) get their own package with a single entry, consumed only via `lazy()`.
- Tests in `__tests__/` next to code; mocks in lowercase `__mocks__/`. No live broker/DB/Redis in unit runs.
- Logging: backend Pino, frontend `frontendLog`. No bare `console.*`.
- Verify from repo root: `npm exec nx -- affected --target=test` and `npm run format:check`. Never run jest/vitest with cwd inside a lib.

`device-keys.ts`: `generateDeviceKeys()` → `{ identityKeyPair (non-extractable), signedPrekeyPair, registrationId }` via `crypto.subtle.generateKey({ name: 'ECDH', namedCurve: 'P-256' }, false, ['deriveKey', 'deriveBits'])`; export only public parts as base64 via `exportPublicKey(key: CryptoKey): Promise<string>` (`exportKey('raw')` → base64) and `importPublicKey(b64: string): Promise<CryptoKey>` (`importKey('raw', bytes, { name: 'ECDH', namedCurve: 'P-256' }, true, [])`.

`ratchet-session.ts` (real X3DH-lite: ephemeral × signed-prekey plus ephemeral × identity — two ECDH shared secrets concatenated, HKDF into the root chain key):

```ts
import type { MessageEnvelope, PrekeyBundleRecord } from '@org/common';

export interface RatchetSession {
  sessionId: string;
  theirDeviceId: string;
  chainKey: CryptoKey;
  messageCounter: number;
}

export async function createSessionFromPrekey(
  theirBundle: PrekeyBundleRecord,
  ourEphemeralPrivate: CryptoKey,
  ourEphemeralPublicB64: string,
): Promise<RatchetSession> {
  const b64 = (s: string) =>
    Uint8Array.from(atob(s), (c) => c.charCodeAt(0));
  const importPub = (s: string) =>
    crypto.subtle.importKey(
      'raw', b64(s),
      { name: 'ECDH', namedCurve: 'P-256' }, true, [],
    );
  const dh1 = await crypto.subtle.deriveBits(
    { name: 'ECDH', public: await importPub(theirBundle.signedPrekey) },
    ourEphemeralPrivate, 256,
  );
  const dh2 = await crypto.subtle.deriveBits(
    { name: 'ECDH', public: await importPub(theirBundle.identityKey) },
    ourEphemeralPrivate, 256,
  );
  const secret = new Uint8Array([...new Uint8Array(dh1), ...new Uint8Array(dh2)]);
  if (theirBundle.oneTimePrekey) {
    const dh3 = await crypto.subtle.deriveBits(
      { name: 'ECDH', public: await importPub(theirBundle.oneTimePrekey) },
      ourEphemeralPrivate, 256,
    );
    secret.set(new Uint8Array(dh3), 64);
  }
  const chainKey = await crypto.subtle.importKey(
    'raw',
    await crypto.subtle.digest('SHA-256', secret),
    { name: 'HKDF' }, false, ['deriveKey'],
  );
  void ourEphemeralPublicB64;
  return {
    sessionId: crypto.randomUUID(),
    theirDeviceId: theirBundle.deviceId,
    chainKey,
    messageCounter: 0,
  };
}

export async function encryptToDevice(
  plaintext: string, session: RatchetSession, senderDeviceId: string,
): Promise<MessageEnvelope> {
  const messageKey = await crypto.subtle.deriveKey(
    {
      name: 'HKDF', hash: 'SHA-256',
      salt: new Uint8Array([session.messageCounter]),
      info: new TextEncoder().encode('msg'),
    },
    session.chainKey,
    { name: 'AES-GCM', length: 256 }, false, ['encrypt', 'decrypt'],
  );
  const iv = crypto.getRandomValues(new Uint8Array(12));
  const ct = await crypto.subtle.encrypt(
    { name: 'AES-GCM', iv }, messageKey, new TextEncoder().encode(plaintext),
  );
  const b64 = (bytes: Uint8Array) =>
    btoa(String.fromCharCode(...bytes));
  const env: MessageEnvelope = {
    senderDeviceId,
    recipientDeviceId: session.theirDeviceId,
    ciphertext: b64(new Uint8Array(ct)),
    iv: b64(iv),
    keyVersion: session.messageCounter,
    ratchetHeader: session.sessionId,
  };
  session.messageCounter += 1;
  return env;
}

export async function decryptFromDevice(
  envelope: MessageEnvelope, session: RatchetSession,
): Promise<string> {
  const messageKey = await crypto.subtle.deriveKey(
    {
      name: 'HKDF', hash: 'SHA-256',
      salt: new Uint8Array([envelope.keyVersion]),
      info: new TextEncoder().encode('msg'),
    },
    session.chainKey,
    { name: 'AES-GCM', length: 256 }, false, ['encrypt', 'decrypt'],
  );
  const b64 = (s: string) =>
    Uint8Array.from(atob(s), (c) => c.charCodeAt(0));
  try {
    const pt = await crypto.subtle.decrypt(
      { name: 'AES-GCM', iv: b64(envelope.iv) },
      messageKey, b64(envelope.ciphertext),
    );
    return new TextDecoder().decode(pt);
  } catch {
    throw new Error('E2EE_DECRYPT_FAILED');
  }
}
```

`decryptFromDevice`: reverse with `keyVersion` as HKDF salt; throws `E2EE_DECRYPT_FAILED` on failure (caller renders placeholder per Task 4 error path).

```ts
// libs/client/entities/message/src/lib/message-idb.ts
import { openDB, type DBSchema } from 'idb';
import type { Message } from '../message.types.js';

interface ChatCacheSchema extends DBSchema {
  messages: { key: string; value: Message & { chatId: string }; indexes: { 'by-chat': string } };
  meta: { key: string; value: { chatId: string; since: string; sinceId?: string } };
}

const dbPromise = openDB<ChatCacheSchema>('polygon-chat-cache', 1, {
  upgrade(db) {
    const messages = db.createObjectStore('messages', { keyPath: 'id' });
    messages.createIndex('by-chat', 'chatId');
    db.createObjectStore('meta', { keyPath: 'chatId' });
  },
});

export async function writeMessagesToCache(
  chatId: string, messages: Message[],
): Promise<void> {
  const db = await dbPromise;
  const tx = db.transaction('messages', 'readwrite');
  await Promise.all(messages.map((m) => tx.store.put({ ...m, chatId })));
  await tx.done;
}

export async function readCachedMessages(
  chatId: string, limit = 200,
): Promise<Message[]> {
  const db = await dbPromise;
  const all = await db.getAllFromIndex('messages', 'by-chat', chatId);
  return all
    .sort((a, b) => a.createdAt.localeCompare(b.createdAt))
    .slice(-limit);
}

export async function evictOldMessages(
  chatId: string, keep = 200,
): Promise<void> {
  const db = await dbPromise;
  const all = await db.getAllFromIndex('messages', 'by-chat', chatId);
  if (all.length <= keep) return;
  const sorted = all.sort((a, b) => a.createdAt.localeCompare(b.createdAt));
  const tx = db.transaction('messages', 'readwrite');
  await Promise.all(
    sorted.slice(0, all.length - keep).map((m) => tx.store.delete(m.id)),
  );
  await tx.done;
}

export async function getLastSync(
  chatId: string,
): Promise<{ since: string; sinceId?: string } | null> {
  const db = await dbPromise;
  const meta = await db.get('meta', chatId);
  return meta ? { since: meta.since, sinceId: meta.sinceId } : null;
}

export async function setLastSync(
  chatId: string, sync: { since: string; sinceId?: string },
): Promise<void> {
  const db = await dbPromise;
  await db.put('meta', { chatId, ...sync });
}
```
